local function CreateButton(name, text, key)
    local Button = Instance.new("TextButton")
    Button.Size = UDim2.new(1, 0, 0, 42)
    Button.BackgroundColor3 = Color3.fromRGB(22, 22, 26) -- Dark Gray-Black Base
    Button.Text = text .. " : [ OFF ]"
    Button.TextColor3 = Color3.fromRGB(180, 180, 180)
    Button.TextSize = 13
    Button.Font = Enum.Font.GothamSemibold
    Button.Parent = Content

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = Button

    local btnStroke = Instance.new("UIStroke")
    btnStroke.Thickness = 1
    btnStroke.Color = Color3.fromRGB(40, 40, 45)
    btnStroke.Parent = Button

    -- Dynamic Custom Theme Toggle Coloring
    local function UpdateColor()
        if _G.PhongHub_Config[key] then
            Button.BackgroundColor3 = Color3.fromRGB(180, 155, 60) -- Dark Blonde Highlight
            Button.TextColor3 = Color3.fromRGB(10, 10, 10) -- Jet Black Text for contrast
            Button.Text = text .. " : [ ON ]"
            btnStroke.Color = Color3.fromRGB(235, 210, 110)
        else
            Button.BackgroundColor3 = Color3.fromRGB(22, 22, 26)
            Button.TextColor3 = Color3.fromRGB(180, 180, 180)
            Button.Text = text .. " : [ OFF ]"
            btnStroke.Color = Color3.fromRGB(40, 40, 45)
        end
    end

    UpdateColor()

    Button.MouseButton1Click:Connect(function()
        _G.PhongHub_Config[key] = not _G.PhongHub_Config[key]
        UpdateColor()
    end)
end

local function SafePhysicalMove(targetPivot)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or not targetPivot then return end
    
    local pos = targetPivot.Position
    local distance = (root.Position - pos).Magnitude
    local duration = distance / _G.PhongHub_Config.MovementSpeed
    
    local info = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(root, info, {CFrame = CFrame.new(pos + Vector3.new(0, 2, 0))})
    tween:Play()
    tween.Completed:Wait()
end

local function FireAllRemotes(name, ...)
    for _, item in pairs(ReplicatedStorage:GetDescendants()) do
        if item:IsA("RemoteEvent") and (item.Name:lower() == name:lower() or item.Name:lower():match(name:lower())) then
            item:FireServer(...)
        end
    end
end

task.spawn(function()
    while true do
        task.wait(1)
        if _G.PhongHub_Config.AutoRaceEvent then
            pcall(function()
                local raceMap = workspace:FindFirstChild("RaceTrack") or workspace:FindFirstChild("RacingEvent") or workspace:FindFirstChild("Race")
                
                FireAllRemotes("JoinRace")
                FireAllRemotes("StartRace")
                FireAllRemotes("EnterRace")
                
                if raceMap then
                    local finishLine = raceMap:FindFirstChild("Finish") or raceMap:FindFirstChild("FinishLine") or raceMap:FindFirstChild("End")
                    if finishLine then
                        SafePhysicalMove(finishLine:GetPivot())
                        task.wait(0.5)
                    else
                        for _, child in pairs(raceMap:GetDescendants()) do
                            if child.Name:lower():match("checkpoint") or child.Name:lower():match("part") then
                                SafePhysicalMove(child:GetPivot())
                                task.wait(0.2)
                            end
                        end
                    end
                end
                
                FireAllRemotes("ClaimRaceRewards")
                FireAllRemotes("ClaimWinPass")
                FireAllRemotes("CollectTrophy")
            end)
        end
    end
end)

        task.wait(0.5)
        if _G.PhongHub_Config.AutoSteal then
            pcall(function()
                local base = workspace:FindFirstChild("PlayerBases") and workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
                local drop = base and (base:FindFirstChild("DropZone") or base:FindFirstChild("EggPen"))
                local folder = workspace:FindFirstChild("Eggs") or workspace:FindFirstChild("EggSpawns") or workspace
                if drop and folder then
                    for _, egg in pairs(folder:GetChildren()) do
                        if not _G.PhongHub_Config.AutoSteal then break end
                        if egg:IsA("Model") and egg.Name:lower():match("egg") and not egg.Name:lower():match("zone") then
                            SafePhysicalMove(egg:GetPivot())
                            task.wait(0.3)
                            SafePhysicalMove(drop:GetPivot())
                            task.wait(0.3)
                        end
                    end
                end
            end)
        end
    end
end)
