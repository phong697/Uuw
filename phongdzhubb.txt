-- ==================== TAB: AUTOMATIONS ====================
local StealToggle = FarmTab:CreateToggle({
   Name = Lang.Text[Lang.Current].toggle_steal,
   CurrentValue = false,
   Callback = function(Value)
      _G.AutoSteal = Value
      if Value then
          task.spawn(function()
              while _G.AutoSteal do
                  for _, obj in ipairs(workspace:GetDescendants()) do
                      if not _G.AutoSteal then break end
                      if obj:IsA("BasePart") and (obj.Name:lower():find("egg") or obj:GetAttribute("Egg")) then
                          local skipEgg = false
                          if _G.AntiBossEvasion then
                              local enemies = workspace:FindFirstChild("Enemies") or workspace:FindFirstChild("Bosses")
                              if enemies then
                                  for _, enemy in ipairs(enemies:GetChildren()) do
                                      local enemyRoot = enemy:FindFirstChild("HumanoidRootPart")
                                      if enemyRoot and (enemyRoot.Position - obj.Position).Magnitude < 15 then
                                          skipEgg = true
                                          break
                                      end
                                  end
                              end
                          end
                          if not skipEgg then
                              local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                              if root then
                                  local distance = (root.Position - (obj.Position + Vector3.new(0, 1.5, 0))).Magnitude
                                  local tweenInfo = TweenInfo.new(distance / _G.TweenSpeed, Enum.EasingStyle.Linear)
                                  local tween = TweenService:Create(root, tweenInfo, {CFrame = obj.CFrame + Vector3.new(0, 1.5, 0)})
                                  tween:Play()
                                  tween.Completed:Wait()
                                  task.wait(math.random(15, 35) / 100)
                              end
                          end
                      end
                  end
                  task.wait(0.5)
              end
          end)
      end
   end,
})

local HatchToggle = FarmTab:CreateToggle({
   Name = Lang.Text[Lang.Current].toggle_hatch,
   CurrentValue = false,
   Callback = function(Value)
      _G.AutoHatch = Value
      if Value then
          task.spawn(function()
              while _G.AutoHatch do
                  local remote = game:GetService("ReplicatedStorage"):FindFirstChild("HatchRemote", true) 
                      or game:GetService("ReplicatedStorage"):FindFirstChild("OpenEgg", true)
                  if remote and remote:IsA("RemoteEvent") then
                      remote:FireServer(_G.SelectedEggType, 1)
                  end
                  task.wait(1.2)
              end
          end)
      end
   end,
})

-- ==================== TAB: EVASION & FILTERS ====================
SettingsTab:CreateToggle({
   Name = Lang.Text[Lang.Current].toggle_boss,
   CurrentValue = false,
   Callback = function(Value)
      _G.AntiBossEvasion = Value
      if Value then
          task.spawn(function()
              while _G.AntiBossEvasion do
                  local character = LocalPlayer.Character
                  local root = character and character:FindFirstChild("HumanoidRootPart")
                  local enemies = workspace:FindFirstChild("Enemies") or workspace:FindFirstChild("Bosses")
                  if root and enemies then
                      for _, enemy in ipairs(enemies:GetChildren()) do
                          local enemyRoot = enemy:FindFirstChild("HumanoidRootPart")
                          if enemyRoot and enemy:FindFirstChild("Humanoid") and enemy.Humanoid.Health > 0 then
                              if (root.Position - enemyRoot.Position).Magnitude < 10 then
                                  root.CFrame = root.CFrame * CFrame.new(0, 8, -5) 
                                  task.wait(0.2)
                              end
                          end
                      end
                  end
                  task.wait(0.1)
              end
          end)
      end
   end,
})

-- Thao tác xác nhận xử lý dọn dẹp sâu đồ họa game
local function ExecuteUltraFix()
local lighting = game:GetService("Lighting")
lighting.GlobalShadows = false
lighting.FogEnd = 9e9
for _, effect in ipairs(lighting:GetDescendants() or lighting:GetChildren()) do
if effect:IsA("PostEffect") or effect:IsA("BloomEffect") or effect:IsA("BlurEffect") or effect:IsA("DepthOfFieldEffect") then
effect.Enabled = false
end
end
for _, obj in ipairs(workspace:GetDescendants()) do
if obj:IsA("Decal") or obj:IsA("Texture") or obj:IsA("Sky") or obj:IsA("Clouds") then
obj:Destroy()
elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
obj.Enabled = false
elseif obj:IsA("BasePart") then
obj.Material = Enum.Material.SmoothPlastic
obj.Reflectance = 0
obj.CastShadow = false
end
end
if settings():GetService("RenderSettings") then
settings():GetService("RenderSettings").QualityLevel = Enum.QualityLevel.Level01
end
Rayfield:Notify({
Title = Lang.Text[Lang.Current].notif_title,
Content = Lang.Text[Lang.Current].notif_heavy,
Duration = 4
})
end
-- Chạy lệnh
ExecuteUltraFix()
end,
