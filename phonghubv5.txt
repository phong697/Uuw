local function CreateTab(tabName, order)
    local TabButton = Instance.new("TextButton")
    TabButton.Size = UDim2.new(1, 0, 0, 40)
    TabButton.BackgroundColor3 = Color3.fromRGB(20, 20, 28)
    TabButton.BorderSizePixel = 0
    TabButton.Text = tabName
    TabButton.TextColor3 = Color3.fromRGB(150, 150, 150)
    TabButton.TextSize = 13
    TabButton.Font = Enum.Font.GothamSemibold
    TabButton.LayoutOrder = order
    TabButton.Parent = TabPanel

    local PageFrame = Instance.new("ScrollingFrame")
    PageFrame.Size = UDim2.new(1, 0, 1, 0)
    PageFrame.BackgroundTransparency = 1
    PageFrame.Visible = false
    PageFrame.CanvasSize = UDim2.new(0, 0, 1, 150)
    PageFrame.ScrollBarThickness = 4
    PageFrame.Parent = PagesContainer

    local PageList = Instance.new("UIListLayout")
    PageList.Padding = UDim.new(0, 8)
    PageList.SortOrder = Enum.SortOrder.LayoutOrder
    PageList.Parent = PageFrame

    TabButton.MouseButton1Click:Connect(function()
        for _, p in pairs(Pages) do p.Visible = false end
        for _, t in pairs(Tabs) do t.TextColor3 = Color3.fromRGB(150, 150, 150) t.BackgroundColor3 = Color3.fromRGB(20, 20, 28) end
        PageFrame.Visible = true
        TabButton.TextColor3 = Color3.fromRGB(0, 180, 255)
        TabButton.BackgroundColor3 = Color3.fromRGB(24, 24, 36)
    end)

    Pages[tabName] = PageFrame
    Tabs[tabName] = TabButton
    return PageFrame
end

        _G.PhongHub_Config[configKey] = not _G.PhongHub_Config[configKey]
        if _G.PhongHub_Config[configKey] then
            Button.BackgroundColor3 = Color3.fromRGB(0, 150, 70)

        else

local function NetworkFire(remoteName, ...)
for _, item in pairs(ReplicatedStorage:GetDescendants()) do
if item:IsA("RemoteEvent") and (item.Name:lower() == remoteName:lower() or item.Name:lower():match(remoteName:lower())) then
item:FireServer(...)
end
end
end
local function LocateSecureDropZone()
local zone = nil
for _, b in pairs(workspace:GetChildren()) do
if b.Name == LocalPlayer.Name or (b:IsA("Model") and b:FindFirstChild("Owner") and b.Owner.Value == LocalPlayer) then
zone = b:FindFirstChild("DropZone") or b:FindFirstChild("EggPen") or b:FindFirstChild("Deposit")
end
end
if not zone and workspace:FindFirstChild("PlayerBases") then
local myBase = workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
if myBase then zone = myBase:FindFirstChild("DropZone") or myBase:FindFirstChild("EggPen") end
end
return zone
end
