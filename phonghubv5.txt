local function CreateTab(tabName, order)
    local TabButton = Instance.new("TextButton")
    TabButton.Name = tabName .. "Tab"
    TabButton.Size = UDim2.new(1, 0, 0, 40)
    TabButton.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
    TabButton.BorderSizePixel = 0
    TabButton.Text = tabName
    TabButton.TextColor3 = Color3.fromRGB(160, 160, 160)
    TabButton.TextSize = 13
    TabButton.Font = Enum.Font.GothamSemibold
    TabButton.LayoutOrder = order
    TabButton.Parent = TabPanel

    local PageFrame = Instance.new("ScrollingFrame")
    PageFrame.Name = tabName .. "Page"
    PageFrame.Size = UDim2.new(1, 0, 1, 0)
    PageFrame.BackgroundTransparency = 1
    PageFrame.Visible = false
    PageFrame.CanvasSize = UDim2.new(0, 0, 1, 100)
    PageFrame.ScrollBarThickness = 4
    PageFrame.Parent = PagesContainer

    local PageList = Instance.new("UIListLayout")
    PageList.Padding = UDim.new(0, 8)
    PageList.SortOrder = Enum.SortOrder.LayoutOrder
    PageList.Parent = PageFrame

    TabButton.MouseButton1Click:Connect(function()
        for _, p in pairs(Pages) do p.Visible = false end
        for _, t in pairs(Tabs) do t.TextColor3 = Color3.fromRGB(160, 160, 160) t.BackgroundColor3 = Color3.fromRGB(22, 22, 30) end
        PageFrame.Visible = true
        TabButton.TextColor3 = Color3.fromRGB(0, 180, 255)
        TabButton.BackgroundColor3 = Color3.fromRGB(26, 26, 38)
    end)

    Pages[tabName] = PageFrame
    Tabs[tabName] = TabButton
    return PageFrame
end

-- Functional Components Inside Pages
local function AddToggle(page, text, configKey)
    local ToggleFrame = Instance.new("Frame")
    ToggleFrame.Size = UDim2.new(1, -10, 0, 45)
    ToggleFrame.BackgroundColor3 = Color3.fromRGB(24, 24, 32)
    ToggleFrame.BorderSizePixel = 0
    ToggleFrame.Parent = page

    local tCorner = Instance.new("UICorner")
    tCorner.CornerRadius = UDim.new(0, 6)
    tCorner.Parent = ToggleFrame

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(0.7, 0, 1, 0)
    Label.Position = UDim2.new(0, 10, 0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = text
    Label.TextColor3 = Color3.fromRGB(230, 230, 230)
    Label.TextSize = 13
    Label.Font = Enum.Font.GothamSemibold
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = ToggleFrame

    local Button = Instance.new("TextButton")
    Button.Size = UDim2.new(0.25, 0, 0, 31)
    Button.Position = UDim2.new(0.72, 0, 0.5, -15)
    Button.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    Button.Text = "OFF"
    Button.TextColor3 = Color3.fromRGB(180, 180, 180)
    Button.TextSize = 12
    Button.Font = Enum.Font.GothamBold
    Button.Parent = ToggleFrame

    local bCorner = Instance.new("UICorner")
    bCorner.CornerRadius = UDim.new(0, 4)
    bCorner.Parent = Button

    if _G.PhongHub_Config[configKey] then
        Button.BackgroundColor3 = Color3.fromRGB(0, 150, 70)
        Button.TextColor3 = Color3.fromRGB(255, 255, 255)
        Button.Text = "ON"
    end

    Button.MouseButton1Click:Connect(function()
        _G.PhongHub_Config[configKey] = not _G.PhongHub_Config[configKey]
        if _G.PhongHub_Config[configKey] then
            Button.BackgroundColor3 = Color3.fromRGB(0, 150, 70)
            Button.TextColor3 = Color3.fromRGB(255, 255, 255)
            Button.Text = "ON"
        else
            Button.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
            Button.TextColor3 = Color3.fromRGB(180, 180, 180)
            Button.Text = "OFF"
        end
    end)
end

    OptBtn.Position = UDim2.new(0.52, 0, 0.5, -15)

local function FireRemotes(name, ...)
    for _, instance in pairs(ReplicatedStorage:GetDescendants()) do
        if instance:IsA("RemoteEvent") and (instance.Name:lower() == name:lower() or instance.Name:lower():match(name:lower())) then
            instance:FireServer(...)
        end
    end
end

local function GetPlayerDropPen()
    local target = nil
for _, b in pairs(workspace:GetChildren()) do
if b.Name == LocalPlayer.Name or (b:IsA("Model") and b:FindFirstChild("Owner") and b.Owner.Value == LocalPlayer) then
target = b:FindFirstChild("DropZone") or b:FindFirstChild("EggPen") or b:FindFirstChild("Deposit")
end
end
if not target and workspace:FindFirstChild("PlayerBases") then
local b = workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
if b then target = b:FindFirstChild("DropZone") or b:FindFirstChild("EggPen") end
end
return target
end
-- Background Processing Module (Anti-Crash Sandbox Loops)
task.spawn(function()
while task.wait(0.3) do
if _G.PhongHub_Config.AutoSteal then
pcall(function()
local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
local zone = GetPlayerDropPen()
local container = workspace:FindFirstChild("Eggs") or workspace:FindFirstChild("EggSpawns") or workspace
if root and zone then
for _, egg in pairs(container:GetChildren()) do
if not _G.PhongHub_Config.AutoSteal then break end
if egg:IsA("Model") and egg.Name:lower():match("egg") and not egg.Name:lower():match("zone") then
root.CFrame = egg:GetPivot()
task.wait(0.15)
root.CFrame = zone:GetPivot()
task.wait(0.15)
end
end
end
end)
end
end
end)
task.spawn(function()
while task.wait(0.1) do
if _G.PhongHub_Config.AutoTreadmill then
pcall(function()
FireRemotes("Train", true)
FireRemotes("Click")
FireRemotes("GainSpeed")
end)
end
end
end)
task.spawn(function()
while task.wait(0.5) do
if _G.PhongHub_Config.AutoHatch then
pcall(function()
-- Dynamic remote calling matching the selected string criteria
FireRemotes("Hatch", _G.PhongHub_Config.SelectedEggType)
FireRemotes("BuyEgg", _G.PhongHub_Config.SelectedEggType)
FireRemotes("OpenEgg", _G.PhongHub_Config.SelectedEggType)
end)
end
end
end)
task.spawn(function()
while task.wait(1.5) do
if _G.PhongHub_Config.AutoUpgradeTreadmill or _G.PhongHub_Config.AutoUpgradeBase then
pcall(function()
if _G.PhongHub_Config.AutoUpgradeTreadmill then
FireRemotes("UpgradeTreadmill")
FireRemotes("BuyTreadmill")
end
if _G.PhongHub_Config.AutoUpgradeBase then
FireRemotes("UpgradeBase")
FireRemotes("UpgradePen")
end
end)
end
end
end)
task.spawn(function()
while task.wait(0.6) do
if _G.PhongHub_Config.AutoEvent then
pcall(function()
local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
if root then
for _, element in pairs(workspace:GetDescendants()) do
if not _G.PhongHub_Config.AutoEvent then break end
local lowName = element.Name:lower()
if (element:IsA("Model") or element:IsA("BasePart")) and (lowName:match("drone") or lowName:match("scramble") or lowName:match("token") or lowName:match("gift")) then
root.CFrame = element:GetPivot()
task.wait(0.2)
end
end
end
FireRemotes("ClaimEventReward")
FireRemotes("CollectPart")
end)
end
end
end)
