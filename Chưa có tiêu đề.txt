-- Functional Components Generator
local function CreateButton(name, text, callback)
    local Button = Instance.new("TextButton")
    Button.Name = name
    Button.Size = UDim2.new(1, 0, 0, 45)
    Button.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
    Button.BorderSizePixel = 0
    Button.Text = text .. " : [ OFF ]"
    Button.TextColor3 = Color3.fromRGB(200, 200, 200)
    Button.TextSize = 14
    Button.Font = Enum.Font.GothamSemibold
    Button.Parent = Content

    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 6)
    BtnCorner.Parent = Button

    local BtnStroke = Instance.new("UIStroke")
    BtnStroke.Thickness = 1
    BtnStroke.Color = Color3.fromRGB(45, 45, 60)
    BtnStroke.Parent = Button

    local toggled = false
    Button.MouseButton1Click:Connect(function()
        toggled = not toggled
        if toggled then
            Button.BackgroundColor3 = Color3.fromRGB(0, 60, 120)
            Button.TextColor3 = Color3.fromRGB(255, 255, 255)
            Button.Text = text .. " : [ ON ]"
            BtnStroke.Color = Color3.fromRGB(0, 180, 255)
        else
            Button.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
            Button.TextColor3 = Color3.fromRGB(200, 200, 200)
            Button.Text = text .. " : [ OFF ]"
            BtnStroke.Color = Color3.fromRGB(45, 45, 60)
        end
        callback(toggled)
    end)
end

-- Safe Remote Invocation
local function SafeInvoke(path, type, ...)
    local success, remote = pcall(function()
        return path:FindFirstChildWhichIsA(type)
    end)
    if success and remote then
        if type == "RemoteEvent" then
            remote:FireServer(...)
        elseif type == "RemoteFunction" then
            remote:InvokeServer(...)
        end
    end
end

-- 1. Auto Steal Mechanism
CreateButton("AutoStealBtn", "Auto Steal Eggs", function(state)
    _G.AutoSteal = state
    if _G.AutoSteal then
        task.spawn(function()
            while _G.AutoSteal do
                pcall(function()
                    local WorkspaceEggs = workspace:FindFirstChild("Eggs") or workspace:FindFirstChild("EggSpawns")
                    local Base = workspace:FindFirstChild("PlayerBases") and workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
                    
                    if WorkspaceEggs and Base then
                        for _, egg in pairs(WorkspaceEggs:GetChildren()) do
                            if _G.AutoSteal == false then break end
                            if egg:IsA("Model") and egg:FindFirstChild("TouchInterest") or egg:FindFirstChild("MeshPart") then
                                if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                                    LocalPlayer.Character.HumanoidRootPart.CFrame = egg:GetPivot()
                                    task.wait(0.2)
                                    
                                    if Base:FindFirstChild("DropZone") or Base:FindFirstChild("EggPen") then
                                        local dropTarget = Base:FindFirstChild("DropZone") or Base:FindFirstChild("EggPen")
                                        LocalPlayer.Character.HumanoidRootPart.CFrame = dropTarget:GetPivot()
                                        task.wait(0.3)
                                    end
                                end
                            end
                        end
                    end
                end)
                task.wait(0.5)
            end
        end)
    end
end)

-- 2. Auto Treadmill Mechanism
CreateButton("AutoTreadmillBtn", "Auto Treadmill Speed", function(state)
    _G.AutoTreadmill = state
    if _G.AutoTreadmill then
        task.spawn(function()
            while _G.AutoTreadmill do
                pcall(function()
                    local Base = workspace:FindFirstChild("PlayerBases") and workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
                    if Base and Base:FindFirstChild("Treadmills") then
                        for _, treadmill in pairs(Base.Treadmills:GetChildren()) do
                            if _G.AutoTreadmill == false then break end
                            if treadmill:FindFirstChildWhichIsA("ProximityPrompt") then
                                fireproximityprompt(treadmill:FindFirstChildWhichIsA("ProximityPrompt"))
                            elseif LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                                LocalPlayer.Character.HumanoidRootPart.CFrame = treadmill:GetPivot()
                            end
                        end
                    end
                    SafeInvoke(ReplicatedStorage, "RemoteEvent", "Train", true)
                end)
                task.wait(0.2)
            end
        end)
    end
end)

-- 3. Auto Event Tracker
CreateButton("AutoEventBtn", "Auto Latest Events (Dr. Scramble)", function(state)
    _G.AutoEvent = state
    if _G.AutoEvent then
        task.spawn(function()
            while _G.AutoEvent do
                pcall(function()
                    local EventFolder = workspace:FindFirstChild("Events") or workspace:FindFirstChild("DrScramble") or workspace
                    for _, obj in pairs(EventFolder:GetChildren()) do
                        if _G.AutoEvent == false then break end
                        if obj.Name:match("Drone") or obj.Name:match("Part") or obj.Name:match("Token") or obj.Name:match("Scramble") then
                            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                                LocalPlayer.Character.HumanoidRootPart.CFrame = obj:GetPivot()
                                task.wait(0.2)
                            end
                        end
                    end
                    SafeInvoke(ReplicatedStorage, "RemoteEvent", "ClaimEventReward")
                    SafeInvoke(ReplicatedStorage, "RemoteEvent", "CollectPart")
                end)
                task.wait(1)
            end
        end)
    end
end)
