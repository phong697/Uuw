local function SafePhysicalMove(targetPivot)
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root or not targetPivot then return end
    
    local pos = targetPivot.Position
    local distance = (root.Position - pos).Magnitude
    local duration = distance / _G.PhongHub_Config.MovementSpeed
    
    local info = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(root, info, {CFrame = CFrame.new(pos + Vector3.new(0, 2, 0))})
    tween:Play()
    tween.Completed:Wait()
end

local function RemoteCall(name, ...)
    for _, item in pairs(ReplicatedStorage:GetDescendants()) do
        if item:IsA("RemoteEvent") and (item.Name:lower() == name:lower() or item.Name:lower():match(name:lower())) then
            item:FireServer(...)
        end
    end
end

local function GetPlayerDropZone()
    local zone = nil
    for _, folder in pairs(workspace:GetChildren()) do
        if folder.Name == LocalPlayer.Name or (folder:IsA("Model") and folder:FindFirstChild("Owner") and folder.Owner.Value == LocalPlayer) then
            zone = folder:FindFirstChild("DropZone") or folder:FindFirstChild("EggPen")
        end
    end
    if not zone and workspace:FindFirstChild("PlayerBases") then
        local myBase = workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
        if myBase then zone = myBase:FindFirstChild("DropZone") or myBase:FindFirstChild("EggPen") end
    end
    return zone
end

-- 1. Stealing Pipeline
task.spawn(function()
    while true do
        task.wait(0.5)
        if _G.PhongHub_Config.AutoSteal then
            pcall(function()
                local drop = GetPlayerDropZone()
                local container = workspace:FindFirstChild("Eggs") or workspace:FindFirstChild("EggSpawns") or workspace
                if drop and container then
                    for _, egg in pairs(container:GetChildren()) do
                        if not _G.PhongHub_Config.AutoSteal then break end
                        if egg:IsA("Model") and egg.Name:lower():match("egg") and not egg.Name:lower():match("zone") then
                            SafePhysicalMove(egg:GetPivot())
                            task.wait(0.3)
                            SafePhysicalMove(drop:GetPivot())
                            task.wait(0.3)
                        end
                    end
                end
            end)
        end
    end
end)

-- 2. Training Pipeline
task.spawn(function()
    while true do
        task.wait(0.2)
        if _G.PhongHub_Config.AutoTreadmill then
            pcall(function()
                RemoteCall("Train", true)
                RemoteCall("Click")
                RemoteCall("GainSpeed")
            end)
        end
    end
end)

-- 3. Hatching Pipeline
task.spawn(function()
    while true do
        task.wait(0.5)
        if _G.PhongHub_Config.AutoHatch then
            pcall(function()
                RemoteCall("Hatch", _G.PhongHub_Config.SelectedEgg)
                RemoteCall("BuyEgg", _G.PhongHub_Config.SelectedEgg)
            end)
        end
    end
end)

-- 4. Event Track Pipeline (Racing Update Core Integration)
task.spawn(function()
    while true do
        task.wait(1)
        if _G.PhongHub_Config.AutoEvent then
            pcall(function()
                RemoteCall("JoinRace")
                RemoteCall("StartRace")
                local track = workspace:FindFirstChild("RaceTrack") or workspace:FindFirstChild("RacingEvent") or workspace:FindFirstChild("Race")
                if track then
                    local finish = track:FindFirstChild("Finish") or track:FindFirstChild("FinishLine")
                    if finish then
                        SafePhysicalMove(finish:GetPivot())
                        task.wait(0.5)
                    end
                end
                RemoteCall("ClaimRaceRewards")
                RemoteCall("CollectTrophy")
            end)
        end
    end
end)
