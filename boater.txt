local function CreateButton(name, text, key)
    local Button = Instance.new("TextButton")

    Button.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
    Button.BorderSizePixel = 0
    Button.Text = text .. " : [ OFF ]"

    Button.TextSize = 14
    Button.Font = Enum.Font.GothamSemibold
    Button.Parent = Content

    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 6)
    BtnCorner.Parent = Button

    local BtnStroke = Instance.new("UIStroke")
    BtnStroke.Thickness = 1
    BtnStroke.Color = Color3.fromRGB(45, 45, 60)
    BtnStroke.Parent = Button

    Button.MouseButton1Click:Connect(function()
        _G.PhongHub_Config[key] = not _G.PhongHub_Config[key]
        if _G.PhongHub_Config[key] then
            Button.BackgroundColor3 = Color3.fromRGB(0, 120, 60)
            Button.TextColor3 = Color3.fromRGB(255, 255, 255)
            Button.Text = text .. " : [ ON ]"
            BtnStroke.Color = Color3.fromRGB(0, 255, 150)
        else
            Button.BackgroundColor3 = Color3.fromRGB(28, 28, 38)
            Button.TextColor3 = Color3.fromRGB(200, 200, 200)
            Button.Text = text .. " : [ OFF ]"
            BtnStroke.Color = Color3.fromRGB(45, 45, 60)
        end
    end)
end

-- Direct Targeting Engine instead of heavy scanning
local function GetNetworkRemote(name)
    local remote = ReplicatedStorage:FindFirstChild(name) or ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild(name)
    if not remote then
        for _, v in pairs(ReplicatedStorage:GetChildren()) do
            if v.Name == name then return v end
        end
    end
    return remote
end

-- 1. Auto Steal Execution Thread
task.spawn(function()
    while task.wait(0.4) do
        if _G.PhongHub_Config.AutoSteal then
            pcall(function()
                local character = LocalPlayer.Character
                local root = character and character:FindFirstChild("HumanoidRootPart")
                local base = workspace:FindFirstChild("PlayerBases") and workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
                local dropZone = base and (base:FindFirstChild("DropZone") or base:FindFirstChild("EggPen"))
                
                -- Fast-check optimized directories
                local eggFolder = workspace:FindFirstChild("Eggs") or workspace:FindFirstChild("EggSpawns") or workspace
                if root and dropZone and eggFolder then
                    for _, obj in pairs(eggFolder:GetChildren()) do
                        if not _G.PhongHub_Config.AutoSteal then break end
                        if obj.Name:lower():match("egg") and obj:IsA("Model") then
                            root.CFrame = obj:GetPivot()
                            task.wait(0.2)
                            root.CFrame = dropZone:GetPivot()
                            task.wait(0.2)
                        end
                    end
                end
            end)
        end
    end
end)

-- 2. Auto Treadmill Execution Thread
task.spawn(function()
    while task.wait(0.2) do
        if _G.PhongHub_Config.AutoTreadmill then
            pcall(function()
                local base = workspace:FindFirstChild("PlayerBases") and workspace.PlayerBases:FindFirstChild(LocalPlayer.Name)
                if base and base:FindFirstChild("Treadmills") then
                    for _, treadmill in pairs(base.Treadmills:GetChildren()) do
                        if not _G.PhongHub_Config.AutoTreadmill then break end
                        local prompt = treadmill:FindFirstChildWhichIsA("ProximityPrompt")
                        if prompt then
                            -- Delta Environment Check for Prompt Execution
                            if fireproximityprompt then
                                fireproximityprompt(prompt)
                            else
                                local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                                if root then root.CFrame = treadmill:GetPivot() end
                            end
                        end
                    end
                end
                
                -- Network Triggers
                local trainRemote = GetNetworkRemote("Train")
                if trainRemote and trainRemote:IsA("RemoteEvent") then trainRemote:FireServer(true) end
                
                local clickRemote = GetNetworkRemote("Click")
                if clickRemote and clickRemote:IsA("RemoteEvent") then clickRemote:FireServer() end
            end)
        end
    end
end)

-- 3. Auto Event Execution Thread (Dr. Scramble Compatibility)
task.spawn(function()
    while task.wait(0.8) do
        if _G.PhongHub_Config.AutoEvent then
            pcall(function()
                local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if root then
                    local targetFolder = workspace:FindFirstChild("Events") or workspace:FindFirstChild("DrScramble") or workspace
                    for _, item in pairs(targetFolder:GetChildren()) do
                        if not _G.PhongHub_Config.AutoEvent then break end
                        local name = item.Name:lower()
                        if name:match("drone") or name:match("part") or name:match("scramble") or name:match("token") then
                            root.CFrame = item:GetPivot()
                            task.wait(0.2)
                        end
                    end
                end
                
                local claimEvent = GetNetworkRemote("ClaimEventReward")
                if claimEvent then claimEvent:FireServer() end
                
                local collectPart = GetNetworkRemote("CollectPart")
                if collectPart then collectPart:FireServer() end
            end)
        end
    end
end)
